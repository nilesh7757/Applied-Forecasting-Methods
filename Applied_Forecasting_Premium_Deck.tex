\sectiontitle{I}{Introduction \& Context}
\conceptslide{The Proactive Mental Health Challenge}{Forecasting vs Detection}{
    \item Most wearable health monitors are \textbf{reactive}.
    \item They notify the user when stress is \textbf{already high}.
    \item \textbf{Our Goal:} A proactive system that predicts increasing stress.
    \item This enables preventative measures like controlled breathing or rest.
}
\conceptslide{Research Scope}{High-Resolution Human Stress}{
    \item \textbf{Dataset:} WESAD (Wearable Stress and Affect Detection).
    \item \textbf{Sensors:} Multimodal (EDA, HR, TEMP).
    \item \textbf{Granularity:} 1Hz resolution (per-second forecasting).
    \item \textbf{Target:} Next-observed continuous stress intensity.
}
\conceptslide{The Impact of Mental Health Forecasting}{Real-World Applications}{
    \item \textbf{Clinical Support:} Monitoring patients with anxiety disorders.
    \item \textbf{Occupational Safety:} Early warning for high-risk professions.
    \item \textbf{Personal Wellness:} Integration into consumer smartwatches.
}

% ==========================================
% PART 3: PIPELINE DEBUGGING
% ==========================================
\sectiontitle{III}{Pipeline Debugging: Fixing Critical Errors}
\conceptslide{The Binary Box Problem}{The Issue with Discrete Labels}{
    \item WESAD provides labels as 0 (Baseline) or 1 (Stress).
    \item \textbf{Forecasting Failure:} Plotting raw labels creates vertical "brick walls".
    \item Linear models cannot learn from perfectly flat 20-minute rectangles.
    \item \textbf{The Fix:} We created a \textbf{Continuous Proxy} reflecting arousal intensity.
}
\conceptslide{The Systemic Data Leakage Crisis}{Identifying the Cheat}{
    \item Early models used Heart Rate at time $t$ to predict Stress at time $t$.
    \item Since HR and Stress occur simultaneously, the model was just "looking up" the label.
    \item \textbf{The Fix:} Strict \textbf{$t-1$ Lag Policy} for all sensor predictors.
}
\imageslide{Visualizing Stationarity}{EDA_Graphs/05_Differenced_Signal_Timeline.png}{Differenced stress signal ($d=1$) oscillating stably around a zero mean.}

% ==========================================
% PART 4: METHODOLOGY
% ==========================================
\sectiontitle{IV}{Forecasting Methodology}
\conceptslide{Statistical Memory Audit}{Determining Optimal Lags}{
    \item We used ACF and PACF plots to identify signal "memory".
    \item \textbf{PACF Result:} Lag-2 significance confirmed the AR(2) requirement.
    \item \textbf{ACF Result:} Lag-3 significance confirmed the MA(3) requirement.
}
\imageslide{Lag Diagnostics}{EDA_Graphs/06_Stationary_Lag_Diagnostics.png}{ACF and PACF evidence for (p,q) parameter selection.}
\conceptslide{The Walk-Forward Protocol}{Simulating Real-World Deployment}{
    \item Traditional 80/20 splits are invalid for high-frequency biosignals.
    \item We use \textbf{Rolling Forecasts}: Predict $t+1$, observe $t+1$, append to history, predict $t+2$.
}

% ==========================================
% PART 5: THE 13 MODELS
% ==========================================
\sectiontitle{V}{The 13 Models Deep-Dive}

% M1
\modeltheoryslide{Model 1: Naive Persistence}{Assumes today is exactly like tomorrow. The baseline for all forecasting.}{\[ \hat{y}_{t+1} = y_t \]}{Determines if advanced complexity is actually providing predictive value.}
\modelresultslide{Naive Persistence}{Forecasting_Graphs/00_Naive_(Baseline).png}{0.0211}{0.0345}{8.68}{0.0673}

% M2
\modeltheoryslide{Model 2: Simple Moving Average}{Averages the last 10 seconds of history to filter out sensor jitter.}{\[ \hat{y}_{t+1} = \frac{1}{10} \sum_{i=0}^9 y_{t-i} \]}{Tests if smoothing the immediate history improves forecast stability.}
\modelresultslide{SMA-10}{Forecasting_Graphs/00_SMA_(Window=10).png}{0.0229}{0.0326}{9.62}{0.1716}

% M3
\modeltheoryslide{Model 3: AutoRegressive (AR)}{Models the target as a linear function of its own lagged values.}{\[ y_t = c + \phi_1 y_{t-1} + \phi_2 y_{t-2} + \epsilon_t \]}{Captures the physiological momentum of the stress response.}
\modelresultslide{AR(2)}{Forecasting_Graphs/03_AR_Optimized.png}{0.0210}{0.0342}{8.66}{0.0840}

% M4
\modeltheoryslide{Model 4: Moving Average (MA)}{Models the next value based on past forecast errors.}{\[ y_t = \mu + \epsilon_t + \theta_1 \epsilon_{t-1} + \theta_2 \epsilon_{t-2} + \theta_3 \epsilon_{t-3} \]}{Allows the system to rapidly correct itself after sudden sensor spikes.}
\modelresultslide{MA(3)}{Forecasting_Graphs/04_MA_Optimized.png}{0.0209}{0.0335}{8.72}{0.1236}

% M5
\modeltheoryslide{Model 5: ARMA(2,3)}{An optimized hybrid of AR and MA components.}{\[ y_t = c + \sum \phi_i y_{t-i} + \epsilon_t + \sum \theta_j \epsilon_{t-j} \]}{Represents the theoretical peak of linear univariate stress forecasting.}
\modelresultslide{ARMA(2,3)}{Forecasting_Graphs/05_ARMA_Optimized.png}{0.0206}{0.0316}{8.60}{0.2204}

% M6
\modeltheoryslide{Model 6: Seasonal ARIMA}{Adds seasonal cycles to the ARIMA equation. We used $s=7$.}{\[ \text{SARIMA}(2,1,3)(1,1,1)_7 \]}{Attempts to capture high-frequency periodicities in physiological stress.}
\modelresultslide{SARIMA}{Forecasting_Graphs/06_SARIMA_Optimized.png}{0.0230}{0.0348}{9.55}{0.0537}

% M7
\modeltheoryslide{Model 7: SARIMAX}{Incorporates eXogenous physiological predictors directly into the equation.}{\[ \hat{y}_t = \text{SARIMA}_t + \boldsymbol{\beta} \mathbf{X}_{t-1} \]}{Tests if external sensors add value to the statistical linear model.}
\modelresultslide{SARIMAX}{Forecasting_Graphs/07_SARIMAX_Optimized.png}{0.0252}{0.0399}{10.38}{-0.2433}

% M8
\modeltheoryslide{Model 8: Random Forest}{A non-parametric ensemble of 100 decision trees.}{\[ \hat{y}_t = \text{Avg}(T_1 \dots T_{100}) \]}{Robust to noise and handles non-linear interactions between signals.}
\modelresultslide{Random Forest}{Forecasting_Graphs/08_Random_Forest_Forecast.png}{0.0275}{0.0377}{11.70}{-0.1089}

% M9
\modeltheoryslide{Model 9: XGBoost}{Gradient Boosting with sequential tree-error minimization.}{\[ \min \sum L(y_i, \hat{y}_i) + \sum \Omega(f_k) \]}{Industry standard for capturing complex physiological thresholds.}
\modelresultslide{XGBoost}{Forecasting_Graphs/09_XGBoost_Forecast.png}{0.0230}{0.0323}{10.06}{0.1840}

% M10
\modeltheoryslide{Model 10: SVR}{Projects signals into higher dimensions via RBF kernels.}{\[ \min \frac{1}{2}||w||^2 + C \sum (\xi_i + \xi_i^*) \]}{Excellent at finding clean margins for stress onset detection.}
\modelresultslide{SVR (RBF Kernel)}{Forecasting_Graphs/11_SVR_Forecast.png}{0.0208}{0.0314}{8.77}{0.2309}

% M11
\modeltheoryslide{Model 11: LSTM}{Recurrent architecture with Input, Output, and Forget gates.}{\[ f_t = \sigma(W_f \cdot [h_{t-1}, x_t] + b_f) \]}{Designed for temporal memory; captured long-range trends most effectively.}
\modelresultslide{LSTM (Deep Learning)}{Forecasting_Graphs/10_LSTM_Forecast.png}{0.0198}{0.0298}{8.40}{0.3057}

% M12
\modeltheoryslide{Model 12: GRU}{Streamlined version of LSTM with only Update/Reset gates.}{\[ z_t = \sigma(W_z \cdot [h_{t-1}, x_t]) \]}{Highly efficient for high-frequency signal processing.}
\modelresultslide{GRU (Deep Learning)}{Forecasting_Graphs/12_GRU_Forecast.png}{0.0202}{0.0301}{8.60}{0.2936}

% M13
\modeltheoryslide{Model 13: Quantile Regression}{Optimizes Pinball Loss to predict specific risk percentiles.}{\[ L_q = \max(q(y - \hat{y}), (q-1)(y - \hat{y})) \]}{Vital for safety-critical systems needing "worst-case" forecasts.}
\modelresultslide{Quantile (95th Pctl)}{Forecasting_Graphs/13_Quantile_Risk_Boundaries.png}{0.0241}{N/A}{N/A}{N/A}
